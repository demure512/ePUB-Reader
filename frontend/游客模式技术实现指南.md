# 游客模式技术实现指南

## 📋 实现概览

游客模式通过本地存储技术实现完整的电子书管理功能，无需服务器交互，数据完全保存在用户浏览器中。

## 🏗️ 核心技术栈

### 前端存储技术
- **localStorage**: 存储书籍元数据、阅读进度、用户设置
- **IndexedDB**: 存储电子书文件内容（支持大文件）
- **sessionStorage**: 临时会话数据

### 状态管理
- **Pinia Store**: 统一状态管理，支持双模式切换
- **Composables**: 可复用的业务逻辑

#### 1.2 IndexedDB 文件存储
```javascript
// src/utils/guestFileStorage.js
export class GuestFileStorage {
  constructor() {
    this.dbName = 'GuestEbookLibrary'
    this.version = 1
    this.db = null
  }
  
  async init() {
    return new Promise((resolve, reject) => {
      const request = indexedDB.open(this.dbName, this.version)
      
      request.onerror = () => reject(request.error)
      request.onsuccess = () => {
        this.db = request.result
        resolve(this.db)
      }
      
      request.onupgradeneeded = (event) => {
        const db = event.target.result
        
        // 创建文件存储表
        if (!db.objectStoreNames.contains('files')) {
          const fileStore = db.createObjectStore('files', { keyPath: 'id' })
          fileStore.createIndex('name', 'name', { unique: false })
          fileStore.createIndex('uploadedAt', 'uploadedAt', { unique: false })
        }
        
        // 创建封面图片存储表
        if (!db.objectStoreNames.contains('covers')) {
          const coverStore = db.createObjectStore('covers', { keyPath: 'bookId' })
        }
      }
    })
  }
  
  async saveFile(bookId, file) {
    const transaction = this.db.transaction(['files'], 'readwrite')
    const store = transaction.objectStore('files')
    
    const fileData = {
      id: bookId,
      name: file.name,
      type: file.type,
      size: file.size,
      content: await file.arrayBuffer(),
      uploadedAt: new Date().toISOString()
    }
    
    return new Promise((resolve, reject) => {
      const request = store.put(fileData)
      request.onsuccess = () => resolve(fileData)
      request.onerror = () => reject(request.error)
    })
  }
  
  async getFile(bookId) {
    const transaction = this.db.transaction(['files'], 'readonly')
    const store = transaction.objectStore('files')
    
    return new Promise((resolve, reject) => {
      const request = store.get(bookId)
      request.onsuccess = () => resolve(request.result)
      request.onerror = () => reject(request.error)
    })
  }
  
  async deleteFile(bookId) {
    const transaction = this.db.transaction(['files'], 'readwrite')
    const store = transaction.objectStore('files')
    
    return new Promise((resolve, reject) => {
      const request = store.delete(bookId)
      request.onsuccess = () => resolve()
      request.onerror = () => reject(request.error)
    })
  }
  
  async saveCover(bookId, coverBlob) {
    const transaction = this.db.transaction(['covers'], 'readwrite')
    const store = transaction.objectStore('covers')
    
    const coverData = {
      bookId,
      data: await coverBlob.arrayBuffer(),
      type: coverBlob.type
    }
    
    return new Promise((resolve, reject) => {
      const request = store.put(coverData)
      request.onsuccess = () => resolve()
      request.onerror = () => reject(request.error)
    })
  }
  
  async getCover(bookId) {
    const transaction = this.db.transaction(['covers'], 'readonly')
    const store = transaction.objectStore('covers')
    
    return new Promise((resolve, reject) => {
      const request = store.get(bookId)
      request.onsuccess = () => {
        const result = request.result
        if (result) {
          const blob = new Blob([result.data], { type: result.type })
          resolve(URL.createObjectURL(blob))
        } else {
          resolve(null)
        }
      }
      request.onerror = () => reject(request.error)
    })
  }
}
```

// src/utils/storageAdapter.js
export class StorageAdapter {
  constructor(mode) {
    this.mode = mode // 'cloud' | 'guest'
    this.fileStorage = mode === 'guest' ? new GuestFileStorage() : null
    
    if (this.mode === 'guest') {
      this.initGuestStorage()
    }
  }
  
  async initGuestStorage() {
    if (this.fileStorage) {
      await this.fileStorage.init()
    }
    
    // 初始化 localStorage 数据
    if (!localStorage.getItem('guest_books')) {
      localStorage.setItem('guest_books', JSON.stringify([]))
    }
    if (!localStorage.getItem('guest_progress')) {
      localStorage.setItem('guest_progress', JSON.stringify({}))
    }
    if (!localStorage.getItem('guest_settings')) {
      localStorage.setItem('guest_settings', JSON.stringify({
        theme: 'light',
        fontSize: 16,
        fontFamily: 'serif',
        lineHeight: 1.6,
        pageWidth: 800,
        autoSave: true,
        autoSaveInterval: 30000
      }))
    }
    if (!localStorage.getItem('guest_categories')) {
      localStorage.setItem('guest_categories', JSON.stringify([]))
    }
  }
  
  // 书籍操作
  async getBooks() {
    if (this.mode === 'cloud') {
      // 调用云端API
      const response = await fetch('/api/books', {
        headers: { 'Authorization': `Bearer ${getToken()}` }
      })
      return await response.json()
    } else {
      // 从本地存储获取
      const books = JSON.parse(localStorage.getItem('guest_books') || '[]')
      
      // 为每本书添加封面URL
      for (const book of books) {
        if (this.fileStorage) {
          book.coverImageUrl = await this.fileStorage.getCover(book.id)
        }
      }
      
      return books
    }
  }
  
  async saveBook(bookData, file = null) {
    if (this.mode === 'cloud') {
      // 上传到云端
      const formData = new FormData()
      if (file) formData.append('file', file)
      Object.keys(bookData).forEach(key => {
        formData.append(key, bookData[key])
      })
      
      const response = await fetch('/api/books/upload', {
        method: 'POST',
        headers: { 'Authorization': `Bearer ${getToken()}` },
        body: formData
      })
      return await response.json()
    } else {
      // 保存到本地
      const books = JSON.parse(localStorage.getItem('guest_books') || '[]')
      const bookId = Date.now().toString()
      
      const newBook = {
        ...bookData,
        id: bookId,
        createdAt: new Date().toISOString(),
        updatedAt: new Date().toISOString()
      }
      
      // 保存文件内容
      if (file && this.fileStorage) {
        await this.fileStorage.saveFile(bookId, file)
        
        // 如果是EPUB，尝试提取封面
        if (file.type === 'application/epub+zip' || file.name.endsWith('.epub')) {
          try {
            const cover = await this.extractEpubCover(file)
            if (cover) {
              await this.fileStorage.saveCover(bookId, cover)
            }
          } catch (error) {
            console.warn('封面提取失败:', error)
          }
        }
      }
      
      books.push(newBook)
      localStorage.setItem('guest_books', JSON.stringify(books))
      
      return newBook
    }
  }
  
  async deleteBook(bookId) {
    if (this.mode === 'cloud') {
      const response = await fetch(`/api/books/${bookId}`, {
        method: 'DELETE',
        headers: { 'Authorization': `Bearer ${getToken()}` }
      })
      return response.ok
    } else {
      const books = JSON.parse(localStorage.getItem('guest_books') || '[]')
      const filteredBooks = books.filter(book => book.id !== bookId)
      localStorage.setItem('guest_books', JSON.stringify(filteredBooks))
      
      // 删除文件和封面
      if (this.fileStorage) {
        await this.fileStorage.deleteFile(bookId)
        // 封面会在下次访问时自动清理
      }
      
      // 删除阅读进度
      const progress = JSON.parse(localStorage.getItem('guest_progress') || '{}')
      delete progress[bookId]
      localStorage.setItem('guest_progress', JSON.stringify(progress))
      
      return true
    }
  }
  
  async getBookFile(bookId) {
    if (this.mode === 'cloud') {
      const response = await fetch(`/api/books/${bookId}/file`, {
        headers: { 'Authorization': `Bearer ${getToken()}` }
      })
      return await response.blob()
    } else {
      const fileData = await this.fileStorage.getFile(bookId)
      if (fileData) {
        return new Blob([fileData.content], { type: fileData.type })
      }
      return null
    }
  }
