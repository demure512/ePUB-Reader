# 桌面应用游客模式方案

## 🎯 问题分析

您提出的问题非常关键：**桌面应用在没有网络的情况下，用户如何使用游客功能？**

原方案中提到的"浏览器存储"确实不适用于桌面应用。我们需要重新设计一个适合桌面应用的本地存储方案。

```javascript
// src/utils/pathManager.js
import { app } from 'electron'
import path from 'path'
import fs from 'fs/promises'

export class PathManager {
  constructor() {
    // 获取用户数据目录
    this.userDataPath = app.getPath('userData')
    this.appDataPath = path.join(this.userDataPath, 'EbookLibrary')
    this.guestDataPath = path.join(this.appDataPath, 'guest')
    
    this.initPaths()
  }
  
  async initPaths() {
    // 创建必要的目录
    const dirs = [
      this.appDataPath,
      this.guestDataPath,
      path.join(this.guestDataPath, 'books'),
      path.join(this.guestDataPath, 'books', 'covers'),
      path.join(this.guestDataPath, 'data'),
      path.join(this.guestDataPath, 'cache'),
      path.join(this.guestDataPath, 'cache', 'thumbnails')
    ]
    
    for (const dir of dirs) {
      try {
        await fs.mkdir(dir, { recursive: true })
      } catch (error) {
        console.error(`创建目录失败: ${dir}`, error)
      }
    }
  }
  
  // 获取各种路径
  getBooksPath() {
    return path.join(this.guestDataPath, 'books')
  }
  
  getCoversPath() {
    return path.join(this.guestDataPath, 'books', 'covers')
  }
  
  getDataPath() {
    return path.join(this.guestDataPath, 'data')
  }
  
  getCachePath() {
    return path.join(this.guestDataPath, 'cache')
  }
  
  // 获取具体文件路径
  getBooksJsonPath() {
    return path.join(this.getDataPath(), 'books.json')
  }
  
  getProgressJsonPath() {
    return path.join(this.getDataPath(), 'progress.json')
  }
  
  getSettingsJsonPath() {
    return path.join(this.getDataPath(), 'settings.json')
  }
}
```

### 1. 文件系统存储适配器

```javascript
// src/utils/desktopStorageAdapter.js
import fs from 'fs/promises'
import path from 'path'
import { PathManager } from './pathManager.js'

export class DesktopStorageAdapter {
  constructor(mode) {
    this.mode = mode // 'cloud' | 'guest'
    this.pathManager = new PathManager()
  }
  
  // 书籍数据操作
  async getBooks() {
    if (this.mode === 'cloud') {
      // 如果有网络，尝试从云端获取
      try {
        return await this.fetchFromCloud('/api/books')
      } catch (error) {
        // 网络失败时从本地缓存获取
        return await this.getBooksFromCache()
      }
    } else {
      // 游客模式：从本地文件获取
      return await this.getBooksFromLocal()
    }
  }
  
  async getBooksFromLocal() {
    try {
      const booksPath = this.pathManager.getBooksJsonPath()
      const data = await fs.readFile(booksPath, 'utf8')
      const books = JSON.parse(data)
      
      // 为每本书添加本地文件路径
      return books.map(book => ({
        ...book,
        localFilePath: path.join(this.pathManager.getBooksPath(), book.fileName),
        localCoverPath: book.coverImage ? 
          path.join(this.pathManager.getCoversPath(), book.coverImage) : null
      }))
    } catch (error) {
      console.log('首次使用，创建空书籍列表')
      return []
    }
  }
  
  async saveBook(bookData, filePath) {
    if (this.mode === 'cloud') {
      return await this.uploadToCloud(bookData, filePath)
    } else {
      return await this.saveBookToLocal(bookData, filePath)
    }
  }
  
  async saveBookToLocal(bookData, filePath) {
    const bookId = Date.now().toString()
    const fileName = `book_${bookId}${path.extname(filePath)}`
    const targetPath = path.join(this.pathManager.getBooksPath(), fileName)
    
    // 复制文件到本地存储
    await fs.copyFile(filePath, targetPath)
    
    // 创建书籍记录
    const newBook = {
      ...bookData,
      id: bookId,
      fileName: fileName,
      filePath: targetPath,
      fileSize: (await fs.stat(filePath)).size,
      createdAt: new Date().toISOString(),
      updatedAt: new Date().toISOString()
    }
    
    // 如果是EPUB，提取封面
    if (fileName.toLowerCase().endsWith('.epub')) {
      try {
        const coverPath = await this.extractEpubCover(filePath, bookId)
        if (coverPath) {
          newBook.coverImage = path.basename(coverPath)
        }
      } catch (error) {
        console.warn('封面提取失败:', error)
      }
    }
    
    // 保存到books.json
    const books = await this.getBooksFromLocal()
    books.push(newBook)
    await this.saveBooksToLocal(books)
    
    return newBook
  }
  
  async saveBooksToLocal(books) {
    const booksPath = this.pathManager.getBooksJsonPath()
    await fs.writeFile(booksPath, JSON.stringify(books, null, 2), 'utf8')
  }
  
  async deleteBook(bookId) {
    if (this.mode === 'cloud') {
      return await this.deleteFromCloud(bookId)
    } else {
      return await this.deleteBookFromLocal(bookId)
    }
  }
  
  async deleteBookFromLocal(bookId) {
    const books = await this.getBooksFromLocal()
    const bookToDelete = books.find(book => book.id === bookId)
    
    if (bookToDelete) {
      // 删除文件
      try {
        await fs.unlink(bookToDelete.filePath)
      } catch (error) {
        console.warn('删除书籍文件失败:', error)
      }
      
      // 删除封面
      if (bookToDelete.coverImage) {
        try {
          const coverPath = path.join(this.pathManager.getCoversPath(), bookToDelete.coverImage)
          await fs.unlink(coverPath)
        } catch (error) {
          console.warn('删除封面文件失败:', error)
        }
      }
      
      // 从列表中移除
      const updatedBooks = books.filter(book => book.id !== bookId)
      await this.saveBooksToLocal(updatedBooks)
      
      // 删除阅读进度
      await this.deleteProgress(bookId)
      
      return true
    }
    
    return false
  }
  
  // 阅读进度操作
  async getProgress(bookId) {
    if (this.mode === 'cloud') {
      try {
        return await this.fetchFromCloud(`/api/books/${bookId}/progress`)
      } catch (error) {
        return await this.getProgressFromLocal(bookId)
      }
    } else {
      return await this.getProgressFromLocal(bookId)
    }
  }
  
  async getProgressFromLocal(bookId) {
    try {
      const progressPath = this.pathManager.getProgressJsonPath()
      const data = await fs.readFile(progressPath, 'utf8')
      const allProgress = JSON.parse(data)
      return allProgress[bookId] || null
    } catch (error) {
      return null
    }
  }
  
  async saveProgress(bookId, progressData) {
    if (this.mode === 'cloud') {
      try {
        await this.postToCloud(`/api/books/${bookId}/progress`, progressData)
      } catch (error) {
        // 网络失败时保存到本地
        await this.saveProgressToLocal(bookId, progressData)
      }
    } else {
      await this.saveProgressToLocal(bookId, progressData)
    }
  }
  
  async saveProgressToLocal(bookId, progressData) {
    try {
      const progressPath = this.pathManager.getProgressJsonPath()
      let allProgress = {}
      
      try {
        const data = await fs.readFile(progressPath, 'utf8')
        allProgress = JSON.parse(data)
      } catch (error) {
        // 文件不存在，使用空对象
      }
      
      allProgress[bookId] = {
        ...progressData,
        updatedAt: new Date().toISOString()
      }
      
      await fs.writeFile(progressPath, JSON.stringify(allProgress, null, 2), 'utf8')
    } catch (error) {
      console.error('保存阅读进度失败:', error)
    }
  }
  
  // 用户设置操作
  async getSettings() {
    if (this.mode === 'cloud') {
      try {
        return await this.fetchFromCloud('/api/user/settings')
      } catch (error) {
        return await this.getSettingsFromLocal()
      }
    } else {
      return await this.getSettingsFromLocal()
    }
  }
  
  async getSettingsFromLocal() {
    try {
      const settingsPath = this.pathManager.getSettingsJsonPath()
      const data = await fs.readFile(settingsPath, 'utf8')
      return JSON.parse(data)
    } catch (error) {
      // 返回默认设置
      return {
        theme: 'light',
        fontSize: 16,
        fontFamily: 'serif',
        lineHeight: 1.6,
        pageWidth: 800,
        autoSave: true,
        autoSaveInterval: 30000
      }
    }
  }
  
  async saveSettings(settings) {
    if (this.mode === 'cloud') {
      try {
        await this.postToCloud('/api/user/settings', settings)
      } catch (error) {
        await this.saveSettingsToLocal(settings)
      }
    } else {
      await this.saveSettingsToLocal(settings)
    }
  }
  
  async saveSettingsToLocal(settings) {
    try {
      const settingsPath = this.pathManager.getSettingsJsonPath()
      await fs.writeFile(settingsPath, JSON.stringify(settings, null, 2), 'utf8')
    } catch (error) {
      console.error('保存设置失败:', error)
    }
  }
  
  // 网络请求方法（带错误处理）
  async fetchFromCloud(endpoint) {
    // 检查网络连接
    if (!navigator.onLine) {
      throw new Error('无网络连接')
    }
    
    const response = await fetch(`${API_BASE_URL}${endpoint}`, {
      headers: {
        'Authorization': `Bearer ${getToken()}`,
        'Content-Type': 'application/json'
      }
    })
    
    if (!response.ok) {
      throw new Error(`网络请求失败: ${response.status}`)
    }
    
    return await response.json()
  }
}
```

## 🔄 离线优先策略

### 1. 网络状态检测

```javascript
// src/composables/useNetworkStatus.js
import { ref, onMounted, onUnmounted } from 'vue'

export function useNetworkStatus() {
  const isOnline = ref(navigator.onLine)
  const networkType = ref('unknown')
  
  const updateNetworkStatus = () => {
    isOnline.value = navigator.onLine
    
    if ('connection' in navigator) {
      networkType.value = navigator.connection.effectiveType || 'unknown'
    }
  }
  
  onMounted(() => {
    window.addEventListener('online', updateNetworkStatus)
    window.addEventListener('offline', updateNetworkStatus)
    updateNetworkStatus()
  })
  
  onUnmounted(() => {
    window.removeEventListener('online', updateNetworkStatus)
    window.removeEventListener('offline', updateNetworkStatus)
  })
  
  return {
    isOnline,
    networkType
  }
}
```

### 2. 智能同步策略

```javascript
// src/utils/syncManager.js
export class SyncManager {
  constructor(storageAdapter) {
    this.adapter = storageAdapter
    this.syncQueue = []
    this.isOnline = navigator.onLine
    
    this.setupNetworkListeners()
  }
  
  setupNetworkListeners() {
    window.addEventListener('online', () => {
      this.isOnline = true
      this.processSyncQueue()
    })
    
    window.addEventListener('offline', () => {
      this.isOnline = false
    })
  }
  
  // 添加到同步队列
  addToSyncQueue(operation) {
    this.syncQueue.push({
      ...operation,
      timestamp: Date.now()
    })
    
    if (this.isOnline) {
      this.processSyncQueue()
    }
  }
  
  // 处理同步队列
  async processSyncQueue() {
    while (this.syncQueue.length > 0 && this.isOnline) {
      const operation = this.syncQueue.shift()
      
      try {
        await this.executeOperation(operation)
      } catch (error) {
        console.error('同步操作失败:', error)
        // 重新加入队列
        this.syncQueue.unshift(operation)
        break
      }
    }
  }
}
```
